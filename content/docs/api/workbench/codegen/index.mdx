    <PyFunction name="generate" type="(result, handlers, namespace) -> str">
      Return the module source for a parse result.

      `handlers` are the handler methods of the callbacks' `App` subclass,
      and `namespace` holds the names that `Widget(name)` may mount from the app.

      <PySourceCode>
        ```python
        def generate(
            result: Mapping[str, Any],
            handlers: Collection[str],
            namespace: Mapping[str, Any],
        ) -> str:
            """Return the module source for a parse result.

            ``handlers`` are the handler methods of the callbacks' ``App`` subclass,
            and ``namespace`` holds the names that `Widget(name)` may mount from the app.
            """
            for key, label in (("states", "$state variables"), ("queries", "Query"), ("mutations", "Mutation")):
                if result.get(key):
                    raise GenerationError(f"{label} is not supported: {', '.join(result[key])}")

            if result.get("root") is None:
                raise GenerationError("The program has no root component.")

            writer = _Writer(set(handlers), namespace)
            root = writer.element(result["root"])
            footer = f"\n        self.widget = {root}\n        app.set_ui(self)\n"
            imports = [text for module, text in IMPORTS.items() if module in ("al", "ax", "workbench") or module in writer.modules]

            return "".join([DOCSTRING, *imports, CLASS, *writer.lines, *writer.bindings, footer])
        ```
